   183→   183→
   184→   184→// Deterministic anonymous display name derived from an id — stable across
   185→   185→// reloads so anonymous users keep the same alias for a given post/comment.
   186→   186→function anonName(id: string): string {
   187→   187→  let hash = 0
   188→   188→  for (let i = 0; i < id.length; i++) {
   189→   189→    hash = (hash * 31 + id.charCodeAt(i)) >>> 0
   190→   190→  }
   191→   191→  const flower = FLOWERS[hash % FLOWERS.length]
   192→   192→  const num = hash % 100
   193→   193→  return `${flower}_${num}`
   194→   194→}
   195→   195→
   196→   196→function timeAgo(iso: string): string {
   197→   197→  const diff = Date.now() - new Date(iso).getTime()
   198→   198→  const mins = Math.floor(diff / 60_000)
   199→   199→  if (mins < 1) return 'Just now'
   200→   200→  if (mins < 60) return `${mins}m ago`
   201→   201→  const hours = Math.floor(mins / 60)
   202→   202→  if (hours < 24) return `${hours}h ago`
   203→   203→  const days = Math.floor(hours / 24)
   204→   204→  if (days < 7) return `${days}d ago`
   205→   205→  const weeks = Math.floor(days / 7)
   206→   206→  if (weeks < 5) return `${weeks}w ago`
   207→   207→  const months = Math.floor(days / 30)
   208→   208→  return `${months}mo ago`
   209→   209→}

   211→   211→// Raw API post → UI Post
   212→   212→interface ApiPost {
   213→   213→  id: string
   214→   214→  title: string
   215→   215→  content: string
   216→   216→  category: string
   217→   217→  isAnonymous: boolean
   218→   218→  likes: number
   219→   219→  reportedCount: number
   220→   220→  createdAt: string
   221→   221→  comments?: Array<{ id: string }>
   222→   222→  user?: { id: string; name: string | null; avatar: string | null }
   223→   223→}
   224→   224→
   225→   225→function mapApiPost(p: ApiPost, currentUserId: string | undefined, likedIds: Set<string>): Post {
   226→   226→  const category = fromApiCategory[p.category] ?? 'General'
   227→   227→  const isOwn = currentUserId && p.user?.id === currentUserId
   228→   228→  const username = p.isAnonymous
   229→   229→    ? anonName(p.id)
   230→   230→    : isOwn
   231→   231→      ? 'You'
   232→   232→      : (p.user?.name || 'Community member')
   233→   233→  return {
   234→   234→    id: p.id,
   235→   235→    username,
   236→   236→    title: p.title,
   237→   237→    content: p.content,
   238→   238→    category,
   239→   239→    likes: p.likes,
   240→   240→    comments: p.comments?.length ?? 0,
   241→   241→    timeAgo: timeAgo(p.createdAt),
   242→   242→    liked: likedIds.has(p.id),
   243→   243→    isAnonymous: p.isAnonymous,
   244→   244→    isOwn: !!isOwn,
   245→   245→    reportedCount: p.reportedCount ?? 0,
   246→   246→    createdAt: p.createdAt,
   247→   247→  }
   248→   248→}
