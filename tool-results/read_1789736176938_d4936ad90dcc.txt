   128→// ─── Custom Tooltip (defined outside render) ─────────────────────────────────
   129→
   130→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   131→  if (active && payload && payload.length) {
   132→    return (
   133→      <div className="bg-white dark:bg-gray-800 rounded-lg shadow-lg border border-gray-200 dark:border-gray-700 p-3 text-xs">
   134→        <p className="font-semibold text-gray-700 dark:text-gray-300 mb-1">{label}</p>
   135→        {payload.map((entry, index) => (
   136→          <p key={index} style={{ color: entry.color }} className="flex items-center gap-1">
   137→            <span className="w-2 h-2 rounded-full inline-block" style={{ backgroundColor: entry.color }} />
   138→            {entry.name}: {typeof entry.value === 'number' ? entry.value.toFixed(1) : entry.value}
   139→          </p>
   140→        ))}
   141→      </div>
   142→    )
   143→  }
   144→  return null
   145→}
