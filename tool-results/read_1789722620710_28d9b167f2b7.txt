   108→export default function AppShell({ user, onLogout }: AppShellProps) {
   109→  const { activeModule, setActiveModule, sidebarOpen, setSidebarOpen, setPremium, setUserProfile, isPremium } = useAppStore()
   110→  const [searchQuery, setSearchQuery] = useState('')
   111→  const [currentTime, setCurrentTime] = useState('')
   112→  const [tourOpen, setTourOpen] = useState(false)
   113→  const [installable, setInstallable] = useState(false)
   114→
   115→  // Track PWA installability (prompt captured by <PwaRegister />)
   116→  useEffect(() => {
   117→    const onInstallable = (e: Event) => {
   118→      const detail = (e as CustomEvent<{ available: boolean }>).detail
   119→      setInstallable(Boolean(detail?.available))
   120→    }
   121→    const onInstalled = () => setInstallable(false)
   122→    window.addEventListener('chandracycle-installable', onInstallable)
   123→    window.addEventListener('chandracycle-installed', onInstalled)
   124→    window.addEventListener('appinstalled', onInstalled)
   125→    return () => {
   126→      window.removeEventListener('chandracycle-installable', onInstallable)
   127→      window.removeEventListener('chandracycle-installed', onInstalled)
   128→      window.removeEventListener('appinstalled', onInstalled)
   129→    }
   130→  }, [])
   131→
   132→  // Sync auth user into the global store
   133→  useEffect(() => {
   134→    setUserProfile({
   135→      id: user.id,
   136→      name: user.name || user.email.split('@')[0],
   137→      email: user.email,
   138→      avatar: user.avatar || undefined,
   139→      cycleLength: user.cycleLength,
   140→      periodLength: user.periodLength,
   141→      lastPeriodStart: user.lastPeriodStart ?? null,
   142→      provider: user.provider as 'email' | 'google' | 'apple',
   143→    })
   144→  }, [user, setUserProfile])
   145→
   146→  // ── Restore premium status from the server (survives reloads) ────────────
   147→  useEffect(() => {
   148→    if (!user.id) return
   149→    let cancelled = false
   150→    fetch(`/api/subscription?userId=${encodeURIComponent(user.id)}`)
   151→      .then((r) => (r.ok ? r.json() : null))
   152→      .then((d) => {
   153→        // Sync BOTH ways: restore premium after reload and drop it when the
   154→        // subscription was cancelled/expired (e.g. from another device).
   155→        if (!cancelled) setPremium(Boolean(d?.active))
   156→      })
   157→      .catch(() => {
   158→        // best-effort — offline users keep their in-session state
   159→      })
   160→    return () => {
   161→      cancelled = true
   162→    }
   163→  }, [user.id, setPremium])
   164→
   165→  // Clock
   166→  useEffect(() => {
   167→    const updateTime = () => {
   168→      const now = new Date()
   169→      setCurrentTime(now.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }))
   170→    }
   171→    updateTime()
   172→    const interval = setInterval(updateTime, 60000)
   173→    return () => clearInterval(interval)
   174→  }, [])
   175→
   176→  // ── Auto-start first-time onboarding tour ────────────────────────────────
   177→  // Fires once per USER (not per browser) — a brand-new signup always sees
   178→  // the tour even if another account dismissed it on the same device.
   179→  useEffect(() => {
   180→    if (typeof window === 'undefined') return
   181→    const tourKey = getTourSeenKey(user.id)
   182→    let seen = false
   183→    try {
   184→      seen = localStorage.getItem(tourKey) === '1'
   185→      // Also clear any stale legacy global flag so it can't shadow new users.
   186→      if (localStorage.getItem('chandracycle_tour_seen') === '1' && !seen) {
   187→        localStorage.removeItem('chandracycle_tour_seen')
   188→      }
   189→    } catch {
   190→      // localStorage may be disabled (private mode) — skip auto-start.
   191→      seen = true
   192→    }
   193→    if (seen) return
   194→    // 2s delay so the dashboard + lazy module finish rendering and the
   195→    // spotlight targets (data-tour="...") are present in the DOM.
   196→    const timer = setTimeout(() => {
   197→      setTourOpen(true)
   198→    }, 2000)
   199→    return () => clearTimeout(timer)
   200→  }, [user.id])
   201→
   202→  const closeTour = () => {
   203→    setTourOpen(false)
   204→    try {
   205→      localStorage.setItem(getTourSeenKey(user.id), '1')
   206→    } catch {
   207→      // ignore — best-effort persistence
   208→    }
   209→  }
   210→
   211→  const activeItem = navItems.find(item => item.id === activeModule)
   212→
   213→  const handleNavClick = (id: ActiveModule) => {
   214→    setActiveModule(id)
   215→  }
   216→
   217→  const renderModule = () => {
   218→    switch (activeModule) {
   219→      case 'dashboard': return <DashboardModule />
   220→      case 'period': return <PeriodModule />
   221→      case 'hormone': return <HormoneModule />
   222→      case 'symptoms': return <SymptomsModule />
   223→      case 'pcos': return <PCOSModule />
   224→      case 'fertility': return <FertilityModule />
   225→      case 'pregnancy': return <PregnancyModule />
   226→      case 'menopause': return <MenopauseModule />
   227→      case 'coach': return <CoachModule />
   228→      case 'diet': return <DietAdvisorModule />
   229→      case 'doctors': return <DoctorFinderModule />
   230→      case 'mental': return <MentalWellnessModule />
   231→      case 'fitness': return <FitnessModule />
   232→      case 'beauty': return <SkinBeautyModule />
   233→      case 'marketplace': return <MarketplaceModule />
   234→      case 'community': return <CommunityModule />
   235→      case 'reports': return <ReportsModule />
   236→      case 'ai-insights': return <AIInsightsModule />
   237→      case 'premium': return <PremiumModule onSubscribe={() => setPremium(true)} />
   238→      case 'settings': return <SettingsModule />
   239→      default: return <DashboardModule />
   240→    }
   241→  }
   242→
   243→  const getGreeting = () => {
   244→    const hour = new Date().getHours()
   245→    if (hour < 12) return 'Good Morning'
   246→    if (hour < 17) return 'Good Afternoon'
   247→    return 'Good Evening'
   248→  }
   249→
   250→  const displayName = user.name || user.email.split('@')[0]
   251→  const initials = displayName.charAt(0).toUpperCase()
   252→  const firstName = displayName.split(' ')[0]
   253→
   254→  // Desktop: sidebar width + collapse. Mobile uses MobileTopbar + MobileBottomNav.
   255→  const desktopMainMargin = sidebarOpen ? 'lg:ml-64' : 'lg:ml-[72px]'
   256→
   257→  return (
   258→    <TooltipProvider delayDuration={200}>
   259→      <div className="h-dvh flex bg-background overflow-hidden">
   260→        {/* ─── Desktop Sidebar (lg+ only) ────────────────────────────────── */}
   261→        <aside
   262→          className={cn(
   263→            'hidden lg:flex flex-col border-r border-border bg-card transition-all duration-300 ease-in-out shrink-0',
   264→            sidebarOpen ? 'w-64' : 'w-[72px]'
   265→          )}
   266→        >
   267→          {/* Logo Area — Luxury brand */}
   268→          <div className="relative flex items-center gap-3 px-4 py-5 border-b border-border overflow-hidden h-[68px]">
   269→            {/* Decorative gold shimmer */}
   270→            <div className="absolute inset-0 bg-gradient-to-r from-amber-500/5 via-rose-500/5 to-transparent pointer-events-none" />
   271→            <div className="relative flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-amber-400 via-rose-500 to-fuchsia-600 text-white font-bold text-lg shrink-0 shadow-lg shadow-rose-500/30">
   272→              <span className="relative">C</span>
   273→              <span className="absolute -top-0.5 -right-0.5 h-2.5 w-2.5 rounded-full bg-amber-300 ring-2 ring-card" />
   274→            </div>
   275→            <AnimatePresence>
   276→              {sidebarOpen && (
   277→                <motion.div
   278→                  initial={{ opacity: 0, x: -10 }}
   279→                  animate={{ opacity: 1, x: 0 }}
   280→                  exit={{ opacity: 0, x: -10 }}
   281→                  transition={{ duration: 0.2 }}
   282→                  className="relative flex flex-col overflow-hidden"
   283→                >
   284→                  <span className="font-serif text-base tracking-tight bg-gradient-to-r from-amber-600 via-rose-600 to-fuchsia-600 bg-clip-text text-transparent font-bold leading-none">
   285→                    ChandraCycle
   286→                  </span>
   287→                  <span className="text-[9px] uppercase tracking-[0.18em] text-muted-foreground/80 leading-tight mt-0.5 font-medium">
   288→                    AI Health Companion
   289→                  </span>
   290→                </motion.div>
   291→              )}
   292→            </AnimatePresence>
   293→          </div>
   294→
   295→          {/* Navigation — native scroll with custom scrollbar */}
   296→          <nav className="flex-1 overflow-y-auto chandracycle-scroll py-3 px-3 space-y-1">
   297→            {navItems.map((item) => {
   298→              const isActive = activeModule === item.id
   299→              return (
   300→                <Tooltip key={item.id}>
   301→                  <TooltipTrigger asChild>
   302→                    <button
   303→                      onClick={() => handleNavClick(item.id)}
   304→                      data-tour={
   305→                        item.id === 'period'
   306→                          ? 'period'
   307→                          : item.id === 'coach'
   308→                            ? 'coach'
   309→                            : item.id === 'symptoms'
   310→                              ? 'symptoms'
   311→                              : undefined
   312→                      }
   313→                      className={cn(
   314→                        'relative flex items-center gap-3 w-full rounded-xl px-3 py-2.5 text-sm font-medium transition-all duration-200 group',
   315→                        isActive
   316→                          ? 'bg-gradient-to-r from-primary/15 via-primary/10 to-transparent text-primary shadow-sm'
   317→                          : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'
   318→                      )}
   319→                    >
   320→                      <item.icon className={cn('h-5 w-5 shrink-0 transition-transform group-hover:scale-110', isActive ? item.color : '')} />
   321→                      <AnimatePresence>
   322→                        {sidebarOpen && (
   323→                          <motion.span
   324→                            initial={{ opacity: 0 }}
   325→                            animate={{ opacity: 1 }}
   326→                            exit={{ opacity: 0 }}
   327→                            className="truncate"
   328→                          >
   329→                            {item.label}
   330→                          </motion.span>
   331→                        )}
   332→                      </AnimatePresence>
   333→                      {item.badge && sidebarOpen && (
   334→                        <Badge variant="secondary" className="ml-auto text-[10px] px-1.5 py-0 h-5 bg-primary/10 text-primary">
   335→                          {item.badge}
   336→                        </Badge>
   337→                      )}
   338→                      {isActive && (
   339→                        <motion.div
   340→                          layoutId="activeIndicator"
   341→                          className="absolute left-0 top-1/2 -translate-y-1/2 w-1 h-6 rounded-r-full bg-gradient-to-b from-amber-500 to-rose-500"
   342→                          transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   343→                        />
   344→                      )}
   345→                    </button>
   346→                  </TooltipTrigger>
   347→                  {!sidebarOpen && (
   348→                    <TooltipContent side="right" className="font-medium">
   349→                      {item.label}
   350→                    </TooltipContent>
   351→                  )}
   352→                </Tooltip>
   353→              )
   354→            })}
   355→          </nav>
   356→
   357→          {/* Bottom section — install, tour, premium + settings */}
   358→          <div className="border-t border-border p-3 space-y-1.5">
   359→            {installable && (
   360→              <button
   361→                onClick={() => window.dispatchEvent(new CustomEvent('chandracycle-install-request'))}
   362→                aria-label="Install ChandraCycle on this device"
   363→                className="group flex items-center gap-3 w-full rounded-xl px-3 py-2.5 text-sm font-medium text-primary hover:bg-primary/10 transition-colors"
   364→              >
   365→                <Smartphone className="h-5 w-5 shrink-0 transition-transform group-hover:scale-110" />
   366→                {sidebarOpen && (
   367→                  <span className="flex items-center gap-1.5">
   368→                    Install app
   369→                    <Badge variant="secondary" className="text-[9px] h-4 px-1.5 bg-primary/10 text-primary border-0">PWA</Badge>
   370→                  </span>
   371→                )}
   372→              </button>
   373→            )}
   374→            <button
   375→              onClick={() => setTourOpen(true)}
   376→              data-tour="replay"
   377→              aria-label="Take a guided tour"
   378→              className="flex items-center gap-3 w-full rounded-xl px-3 py-2.5 text-sm font-medium text-muted-foreground hover:bg-accent hover:text-accent-foreground transition-colors"
   379→            >
   380→              <HelpCircle className="h-5 w-5 shrink-0" />
   381→              {sidebarOpen && <span>Take Tour</span>}
   382→            </button>
   383→            <button
   384→              onClick={() => handleNavClick('premium')}
   385→              data-tour="premium"
   386→              className="group relative flex items-center gap-3 w-full rounded-xl px-3 py-2.5 text-sm font-medium overflow-hidden bg-gradient-to-r from-amber-500 via-yellow-500 to-orange-500 text-white hover:shadow-lg hover:shadow-amber-500/30 transition-all"
   387→            >
   388→              <span className="absolute inset-0 bg-gradient-to-r from-white/0 via-white/20 to-white/0 -translate-x-full group-hover:translate-x-full transition-transform duration-700" />
   389→              <Crown className="h-5 w-5 shrink-0 relative" />
   390→              {sidebarOpen && (
   391→                <span className="flex items-center gap-1.5 relative">
   392→                  Go Premium
   393→                  <Badge variant="secondary" className="text-[9px] h-4 px-1.5 bg-white/25 text-white border-0">PRO</Badge>
   394→                </span>
   395→              )}
   396→            </button>
   397→            <button
   398→              onClick={() => handleNavClick('settings')}
   399→              className="flex items-center gap-3 w-full rounded-xl px-3 py-2.5 text-sm text-muted-foreground hover:bg-accent hover:text-accent-foreground transition-colors"
   400→            >
   401→              <Settings className="h-5 w-5 shrink-0" />
   402→              {sidebarOpen && <span>Settings</span>}
   403→            </button>
   404→          </div>
   405→        </aside>
   406→
   407→        {/* ─── Main Content ───────────────────────────────────────────────── */}
   408→        <div className={cn('flex-1 min-w-0 flex flex-col h-full transition-all duration-300 ease-in-out', desktopMainMargin)}>
   409→          {/* Desktop Top Bar (lg+ only) — luxury glass header */}
   410→          <header className="hidden lg:flex shrink-0 z-30 items-center justify-between gap-4 border-b border-border bg-card/70 backdrop-blur-xl px-4 lg:px-5 py-2.5 relative">
   411→            {/* Subtle top gradient line */}
   412→            <div className="absolute top-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-amber-500/40 to-transparent" />
   413→            <div className="flex items-center gap-3 min-w-0 flex-1">
   414→              <Button
   415→                variant="ghost"
   416→                size="icon"
   417→                onClick={() => setSidebarOpen(!sidebarOpen)}
   418→                className="shrink-0"
   419→                aria-label="Toggle menu"
   420→              >
   421→                {sidebarOpen ? <X className="h-4 w-4" /> : <Menu className="h-4 w-4" />}
   422→              </Button>
   423→              <div className="flex items-center gap-2 min-w-0">
   424→                {/* Compact greeting on lg, full greeting on xl */}
   425→                <span className="text-sm font-medium truncate hidden xl:inline">{getGreeting()}, {firstName}</span>
   426→                <span className="text-sm font-medium truncate xl:hidden">Hi, {firstName}</span>
   427→                <ChevronRight className="h-3.5 w-3.5 text-muted-foreground shrink-0 hidden xl:inline" />
   428→                <span className="text-sm text-muted-foreground truncate hidden xl:inline">{activeItem?.label}</span>
   429→              </div>
   430→            </div>
   431→
   432→            <div className="flex items-center gap-3 shrink-0">
   433→              <div className="relative hidden xl:block">
   434→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   435→                <Input
   436→                  placeholder="Search..."
   437→                  value={searchQuery}
   438→                  onChange={(e) => setSearchQuery(e.target.value)}
   439→                  className="pl-9 w-56 h-9 bg-muted/50 border-0 focus-visible:ring-1"
   440→                />
   441→              </div>
   442→              <span className="text-xs text-muted-foreground hidden 2xl:block tabular-nums">{currentTime}</span>
   443→
   444→              {/* Active notification bar */}
   445→              <NotificationPanel userId={user.id} />
   446→
   447→              {/* Profile dropdown with brand presence */}
   448→              <DropdownMenu>
   449→                <DropdownMenuTrigger asChild>
   450→                  <button
