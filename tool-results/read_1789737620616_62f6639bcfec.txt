   267→function PlanCard({ plan, billingCycle, onChoosePlan }: { plan: Plan; billingCycle: BillingCycle; onChoosePlan: (plan: Plan) => void }) {
   268→  const hasPremium = useAppStore((s) => s.hasPremium())
   269→  const price = billingCycle === 'monthly' ? plan.monthlyPrice : plan.yearlyPrice
   270→  const isFree = plan.id === 'free'
   271→  // User is on "free" if they don't have premium; otherwise they're on "premium"
   272→  const isCurrentPlan = hasPremium ? plan.id === 'premium' : plan.id === 'free'
   273→  const yearlyPerMonth = plan.yearlyPrice / 12
   274→  const savings = plan.monthlyPrice > 0 && plan.yearlyPrice > 0
   275→    ? Math.round(((plan.monthlyPrice * 12 - plan.yearlyPrice) / (plan.monthlyPrice * 12)) * 100)
   276→    : 0
   277→
   278→  return (
   279→    <motion.div
   280→      initial={{ opacity: 0, y: 20 }}
   281→      whileInView={{ opacity: 1, y: 0 }}
   282→      viewport={{ once: true }}
   283→      transition={{ duration: 0.4 }}
   284→      whileHover={{ y: -4 }}
   285→      className={cn('relative', plan.isPopular && 'lg:-mt-4 lg:mb-0')}
   286→    >
   287→      <Card className={cn(
   288→        'relative h-full overflow-hidden transition-all',
   289→        plan.isPopular
   290→          ? 'border-amber-400 dark:border-amber-600 shadow-xl shadow-amber-500/20 ring-2 ring-amber-400/50'
   291→          : plan.borderColor
   292→      )}>
   293→        {/* Popular badge */}
   294→        {plan.isPopular && (
   295→          <div className="absolute top-0 left-1/2 -translate-x-1/2 -translate-y-1/2 z-10">
   296→            <div className="flex items-center gap-1.5 rounded-full bg-gradient-to-r from-amber-400 via-yellow-500 to-orange-500 px-4 py-1.5 text-xs font-bold text-white shadow-lg">
   297→              <Sparkles className="h-3 w-3" />
   298→              {plan.badgeText}
   299→            </div>
   300→          </div>
   301→        )}
   302→
   303→        {/* Gradient top */}
   304→        <div className={cn('h-2 bg-gradient-to-r', plan.gradient)} />
   305→
   306→        <CardHeader className="pt-5 pb-3">
   307→          <div className="flex items-center justify-between">
   308→            <div className={cn('flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br text-white shadow-md', plan.gradient)}>
   309→              <plan.icon className="h-5 w-5" />
   310→            </div>
   311→            {savings > 0 && billingCycle === 'yearly' && (
   312→              <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 gap-1">
   313→                <Gift className="h-3 w-3" /> Save {savings}%
   314→              </Badge>
   315→            )}
   316→          </div>
   317→          <div className="mt-2">
   318→            <CardTitle className="text-xl">{plan.name}</CardTitle>
   319→            <CardDescription>{plan.tagline}</CardDescription>
   320→          </div>
   321→        </CardHeader>
   322→
   323→        <CardContent className="pb-3">
   324→          {/* Price */}
   325→          <div className="mb-4">
   326→            {isFree ? (
   327→              <div className="flex items-baseline gap-1">
   328→                <span className="text-4xl font-bold">₹0</span>
   329→                <span className="text-sm text-muted-foreground">forever</span>
   330→              </div>
   331→            ) : (
   332→              <div>
   333→                <div className="flex items-baseline gap-1">
   334→                  <span className="text-4xl font-bold">₹{price.toLocaleString('en-IN')}</span>
   335→                  <span className="text-sm text-muted-foreground">/{billingCycle === 'monthly' ? 'month' : 'year'}</span>
   336→                </div>
   337→                {billingCycle === 'yearly' && (
   338→                  <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1">
   339→                    Just ₹{yearlyPerMonth.toFixed(0)}/month · billed annually
   340→                  </p>
   341→                )}
   342→              </div>
   343→            )}
   344→          </div>
   345→
   346→          <Separator className="mb-4" />
   347→
   348→          {/* Features */}
   349→          <ul className="space-y-2.5">
   350→            {plan.features.map((feature) => (
   351→              <li key={feature.text} className="flex items-start gap-2.5">
   352→                <div className={cn(
   353→                  'flex h-4 w-4 items-center justify-center rounded-full shrink-0 mt-0.5',
   354→                  feature.included
   355→                    ? plan.isPopular ? 'bg-amber-100 dark:bg-amber-950/50' : 'bg-emerald-100 dark:bg-emerald-950/40'
   356→                    : 'bg-muted/50'
   357→                )}>
   358→                  {feature.included ? (
   359→                    <Check className={cn('h-3 w-3', plan.isPopular ? 'text-amber-600 dark:text-amber-400' : 'text-emerald-600 dark:text-emerald-400')} />

   361→                    <X className="h-2.5 w-2.5 text-muted-foreground" />
   362→                  )}
   363→                </div>
   364→                <span className={cn(
   365→                  'text-sm leading-snug',
   366→                  feature.included ? 'text-foreground' : 'text-muted-foreground/60 line-through'
   367→                )}>
   368→                  {feature.text}
   369→                </span>
   370→              </li>
   371→            ))}
   372→          </ul>
   373→        </CardContent>
   374→
   375→        <CardFooter className="pt-0">
   376→          <Button
   377→            className={cn(
   378→              'w-full',
   379→              isCurrentPlan
   380→                ? 'bg-muted text-muted-foreground'
   381→                : plan.isPopular
   382→                ? 'bg-gradient-to-r from-amber-500 via-yellow-500 to-orange-500 hover:opacity-90 text-white shadow-md shadow-amber-500/30'
   383→                : plan.id === 'plus'
   384→                ? 'bg-gradient-to-r from-purple-500 via-fuchsia-500 to-pink-500 hover:opacity-90 text-white shadow-md shadow-fuchsia-500/30'
   385→                : 'bg-foreground text-background hover:bg-foreground/90'
   386→            )}
   387→            disabled={isCurrentPlan}
   388→            onClick={() => {
   389→              if (isCurrentPlan) return
   390→              onChoosePlan(plan)
   391→            }}
   392→          >
   393→            {isCurrentPlan ? (
   394→              <>
   395→                <Check className="h-4 w-4 mr-1.5" /> Current Plan
   396→              </>
   397→            ) : (
   398→              <>
   399→                {plan.cta}
   400→                <ArrowRight className="h-4 w-4 ml-1.5" />
   401→              </>
   402→            )}
   403→          </Button>
   404→        </CardFooter>
   405→      </Card>
   406→    </motion.div>
   407→  )
   408→}
