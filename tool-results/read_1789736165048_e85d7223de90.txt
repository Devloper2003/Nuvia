   263→function correlationColor(value: number): string {
   264→  if (value >= 0.7) return 'bg-emerald-500'
   265→  if (value >= 0.4) return 'bg-emerald-400/70'
   266→  if (value >= 0.2) return 'bg-yellow-400/60'
   267→  if (value > -0.2) return 'bg-slate-300/50'
   268→  if (value > -0.4) return 'bg-orange-400/60'
   269→  if (value > -0.7) return 'bg-rose-400/70'
   270→  return 'bg-rose-500'
   271→}
   272→
   273→function correlationText(value: number): string {
   274→  const abs = Math.abs(value)
   275→  const direction = value > 0 ? 'positive' : value < 0 ? 'negative' : 'neutral'
   276→  let strength = 'weak'
   277→  if (abs >= 0.7) strength = 'very strong'
   278→  else if (abs >= 0.4) strength = 'moderate'
   279→  else if (abs >= 0.2) strength = 'mild'
   280→  return `${strength} ${direction}`
   281→}

   310→// Animated count-up hook
   311→function useCountUp(target: number, duration = 1.8) {
   312→  const [value, setValue] = useState(0)
   313→  useEffect(() => {
   314→    let raf = 0
   315→    const start = performance.now()
   316→    const tick = (now: number) => {
   317→      const t = Math.min(1, (now - start) / (duration * 1000))
   318→      const eased = 1 - Math.pow(1 - t, 3)
   319→      setValue(Math.round(target * eased))
   320→      if (t < 1) raf = requestAnimationFrame(tick)
   321→    }
   322→    raf = requestAnimationFrame(tick)
   323→    return () => cancelAnimationFrame(raf)
   324→  }, [target, duration])
   325→  return value
   326→}

   330→function HealthScoreGauge({ score }: { score: number }) {
   331→  const animatedScore = useCountUp(score, 2)
   332→  const size = 240
   333→  const stroke = 18
   334→  const radius = (size - stroke) / 2
   335→  const circumference = 2 * Math.PI * radius
   336→  const offset = circumference * (1 - animatedScore / 100)
   337→
   338→  return (
   339→    <div className="relative" style={{ width: size, height: size }}>
   340→      <svg width={size} height={size} className="-rotate-90">
   341→        <defs>
   342→          <linearGradient id="scoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   343→            <stop offset="0%" stopColor="#8b5cf6" />
   344→            <stop offset="50%" stopColor="#a855f7" />
   345→            <stop offset="100%" stopColor="#f59e0b" />
   346→          </linearGradient>
   347→          <filter id="glow">
   348→            <feGaussianBlur stdDeviation="3" result="coloredBlur" />
   349→            <feMerge>
   350→              <feMergeNode in="coloredBlur" />
   351→              <feMergeNode in="SourceGraphic" />
   352→            </feMerge>
   353→          </filter>
   354→        </defs>
   355→        {/* Background ring */}
   356→        <circle
   357→          cx={size / 2}
   358→          cy={size / 2}
   359→          r={radius}
   360→          fill="none"
   361→          stroke="oklch(0.91 0.02 325)"
   362→          strokeWidth={stroke}
   363→          className="dark:stroke-white/10"
   364→        />
   365→        {/* Progress arc */}
   366→        <motion.circle
   367→          cx={size / 2}
   368→          cy={size / 2}
   369→          r={radius}
   370→          fill="none"
   371→          stroke="url(#scoreGradient)"
   372→          strokeWidth={stroke}
   373→          strokeLinecap="round"
   374→          strokeDasharray={circumference}
   375→          initial={{ strokeDashoffset: circumference }}
   376→          animate={{ strokeDashoffset: offset }}
   377→          transition={{ duration: 2, ease: 'easeOut' }}
   378→          filter="url(#glow)"

   380→      </svg>
   381→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   382→        <motion.div
   383→          initial={{ opacity: 0, scale: 0.8 }}
   384→          animate={{ opacity: 1, scale: 1 }}

   392→          <Sparkles className="h-3 w-3 mr-1" /> Highly Predictable
   393→        </Badge>
   394→      </div>
   395→    </div>
   396→  )
   397→}
