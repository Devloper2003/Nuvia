   116→
   117→function getPhaseForCycleDay(cycleDay: number, cycleLength: number, periodLength: number) {
   118→  // Period phase
   119→  if (cycleDay <= periodLength) {
   120→    return PHASE_META[0]
   121→  }
   122→  // Ovulation is roughly 14 days before next period
   123→  const ovulationDay = cycleLength - 14
   124→  if (cycleDay >= ovulationDay - 1 && cycleDay <= ovulationDay + 1) {
   125→    return PHASE_META[2]
   126→  }
   127→  if (cycleDay < ovulationDay - 1) {
   128→    return PHASE_META[1]
   129→  }
   130→  return PHASE_META[3]
   131→}

   137→function formatReminderDate(isoDate: string): string {
   138→  const todayIso = new Date().toISOString().split('T')[0]
   139→  const tomorrowIso = new Date(Date.now() + 86_400_000).toISOString().split('T')[0]
   140→  if (isoDate === todayIso) return 'Today'
   141→  if (isoDate === tomorrowIso) return 'Tomorrow'
   142→  const d = new Date(isoDate + 'T00:00:00')
   143→  return d.toLocaleDateString('en-IN', { weekday: 'short', day: 'numeric', month: 'short' })
   144→}

   161→// ─── Hormone educational curves ────────────────────────────────────────────
   162→// Cosine easing over t∈[0,1]
   163→function ease(t: number): number {
   164→  const clamped = Math.max(0, Math.min(1, t))
   165→  return 0.5 - 0.5 * Math.cos(Math.PI * clamped)
   166→}

   168→function buildHormoneCurves(cycleLength: number) {
   169→  const ovul = Math.max(10, cycleLength - 14)
   170→  const points: Array<{ day: number; label: string; estrogen: number; progesterone: number }> = []
   171→  for (let d = 1; d <= cycleLength; d++) {
   172→    // Estrogen: low during period → follicular rise → ovulation peak → dip →
   173→    // secondary luteal bump → premenstrual fall
   174→    let estrogen: number
   175→    if (d <= ovul) {
   176→      estrogen = 15 + 75 * ease((d - 3) / (ovul - 3))
   177→    } else if (d <= ovul + 2) {
   178→      estrogen = 90 - 55 * ease((d - ovul) / 2)
   179→    } else if (d <= ovul + 8) {
   180→      estrogen = 35 + 22 * ease((d - ovul - 2) / 6)
   181→    } else {
   182→      estrogen = 57 - 40 * ease((d - ovul - 8) / Math.max(1, cycleLength - ovul - 8))
   183→    }
   184→    // Progesterone: flat low until ovulation → strong luteal rise → fall
   185→    let progesterone: number
   186→    if (d <= ovul) {
   187→      progesterone = 12
   188→    } else if (d <= ovul + 7) {
   189→      progesterone = 12 + 73 * ease((d - ovul) / 7)
   190→    } else {
   191→      progesterone = 85 - 70 * ease((d - ovul - 7) / Math.max(1, cycleLength - ovul - 7))
   192→    }
   193→    points.push({
   194→      day: d,
   195→      label: d === 1 || d % 7 === 0 || d === cycleLength ? `D${d}` : '',
   196→      estrogen: Math.round(estrogen),
   197→      progesterone: Math.round(progesterone),
   198→    })
   199→  }
   200→  return { points, ovul }
   201→}

  1259→/**
  1260→ * Count-up number animation for dashboard stats. Animates from the previous
  1261→ * value to the new one with an ease-out cubic; renders the final value
  1262→ * instantly when the user prefers reduced motion.
  1263→ */
  1264→function AnimatedNumber({ value, duration = 900 }: { value: number; duration?: number }) {
  1265→  const [display, setDisplay] = useState(0)
  1266→  const prev = useRef(0)
  1267→
  1268→  useEffect(() => {
  1269→    // Reduced motion → jump straight to the value (still via rAF so no
  1270→    // synchronous setState inside the effect body).
  1271→    const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches
  1272→    const from = reduce ? value : prev.current
  1273→    const start = performance.now()
  1274→    const dur = reduce ? 0 : duration
  1275→    let raf = 0
  1276→    const tick = (now: number) => {
  1277→      const t = Math.min((now - start) / dur, 1)
  1278→      const eased = 1 - Math.pow(1 - t, 3)
  1279→      setDisplay(Math.round(from + (value - from) * eased))
  1280→      if (t < 1) {
  1281→        raf = requestAnimationFrame(tick)
  1282→      } else {
  1283→        prev.current = value
  1284→      }
  1285→    }
  1286→    raf = requestAnimationFrame(tick)
  1287→    return () => cancelAnimationFrame(raf)
  1288→  }, [value, duration])
  1289→
  1290→  return <>{display}</>
  1291→}

  1396→function CycleProgressRing({
  1397→  cycleDay,
  1398→  cycleLength,
  1399→  periodLength,
  1400→  phaseColor,
  1401→  phaseName,
  1402→  ofDaysLabel,
  1403→}: {
  1404→  cycleDay: number
  1405→  cycleLength: number
  1406→  periodLength: number
  1407→  phaseColor: string
  1408→  phaseName: string
  1409→  ofDaysLabel: string
  1410→}) {
  1411→  const radius = 90
  1412→  const strokeWidth = 10
  1413→  const normalizedRadius = radius - strokeWidth / 2
  1414→  const circumference = normalizedRadius * 2 * Math.PI
  1415→  const progress = cycleDay / cycleLength
  1416→  const strokeDashoffset = circumference - progress * circumference
  1417→
  1418→  // Phase proportions
  1419→  const menstrualPct = periodLength / cycleLength
  1420→  const ovulationPct = 3 / cycleLength
  1421→  const follicularPct = (cycleLength - 14 - 1 - periodLength) / cycleLength
  1422→  const lutealPct = 1 - menstrualPct - ovulationPct - follicularPct
  1423→
  1424→  return (
  1425→    <div className="relative flex items-center justify-center">
  1426→      <svg
  1427→        height={radius * 2}
  1428→        width={radius * 2}
  1429→        className="transform -rotate-90"
  1430→        role="img"
  1431→        aria-label={`Cycle day ${cycleDay} of ${cycleLength}`}
  1432→      >
  1433→        <circle
  1434→          stroke="oklch(0.91 0.02 325)"
  1435→          fill="transparent"
  1436→          strokeWidth={strokeWidth}
  1437→          r={normalizedRadius}
  1438→          cx={radius}
  1439→          cy={radius}
  1440→        />
  1441→        {/* Menstrual */}
  1442→        <circle
  1443→          stroke="#e11d48"
  1444→          fill="transparent"
  1445→          strokeWidth={strokeWidth}
  1446→          strokeDasharray={`${circumference * menstrualPct} ${circumference}`}
  1447→          strokeDashoffset={0}
  1448→          r={normalizedRadius}
  1449→          cx={radius}
  1450→          cy={radius}
  1451→          opacity={0.3}
  1452→          strokeLinecap="round"
  1453→        />
  1454→        {/* Follicular */}
  1455→        <circle
  1456→          stroke="#8b5cf6"
  1457→          fill="transparent"
  1458→          strokeWidth={strokeWidth}
  1459→          strokeDasharray={`${circumference * follicularPct} ${circumference}`}
  1460→          strokeDashoffset={-circumference * menstrualPct}
  1461→          r={normalizedRadius}
  1462→          cx={radius}
  1463→          cy={radius}
  1464→          opacity={0.3}
  1465→          strokeLinecap="round"
  1466→        />
  1467→        {/* Ovulation */}
  1468→        <circle
  1469→          stroke="#f97316"
  1470→          fill="transparent"
  1471→          strokeWidth={strokeWidth}
  1472→          strokeDasharray={`${circumference * ovulationPct} ${circumference}`}
  1473→          strokeDashoffset={-circumference * (menstrualPct + follicularPct)}
  1474→          r={normalizedRadius}
  1475→          cx={radius}
  1476→          cy={radius}
  1477→          opacity={0.3}
  1478→          strokeLinecap="round"
  1479→        />
  1480→        {/* Luteal */}
  1481→        <circle
  1482→          stroke="#06b6d4"
  1483→          fill="transparent"
  1484→          strokeWidth={strokeWidth}
  1485→          strokeDasharray={`${circumference * lutealPct} ${circumference}`}
  1486→          strokeDashoffset={-circumference * (menstrualPct + follicularPct + ovulationPct)}
  1487→          r={normalizedRadius}
  1488→          cx={radius}
  1489→          cy={radius}
  1490→          opacity={0.3}
  1491→          strokeLinecap="round"
  1492→        />
  1493→        {/* Active progress arc */}
  1494→        <circle
  1495→          stroke={phaseColor}
  1496→          fill="transparent"
  1497→          strokeWidth={strokeWidth + 2}
  1498→          strokeDasharray={`${circumference} ${circumference}`}
  1499→          strokeDashoffset={strokeDashoffset}
  1500→          strokeLinecap="round"
  1501→          r={normalizedRadius}
  1502→          cx={radius}
  1503→          cy={radius}
  1504→          style={{ transition: 'stroke-dashoffset 1.2s ease-in-out' }}
  1505→          className="drop-shadow-lg"
  1506→        />
  1507→      </svg>
  1508→      <div className="absolute inset-0 flex flex-col items-center justify-center">
  1509→        <span className="text-4xl font-bold gradient-text">{cycleDay}</span>
  1510→        <span className="text-xs text-muted-foreground mt-0.5">{ofDaysLabel}</span>
  1511→        <Badge
  1512→          className="mt-2 border-0 text-white font-semibold text-xs px-3 py-1"
  1513→          style={{ backgroundColor: phaseColor }}
  1514→        >
  1515→          {phaseName}
  1516→        </Badge>
  1517→      </div>
  1518→    </div>
  1519→  )
  1520→}
