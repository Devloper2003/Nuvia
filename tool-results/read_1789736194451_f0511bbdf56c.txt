   306→function FaceDiagram({ selected, onToggle }: { selected: FaceArea[]; onToggle: (area: FaceArea) => void }) {
   307→  const isSelected = (area: FaceArea) => selected.includes(area)
   308→
   309→  const areas: Record<FaceArea, { cx: number; cy: number; r: number; label: string }> = {
   310→    forehead: { cx: 100, cy: 38, r: 26, label: 'Forehead' },
   311→    tzone: { cx: 100, cy: 70, r: 14, label: 'T-Zone' },
   312→    nose: { cx: 100, cy: 88, r: 10, label: 'Nose' },
   313→    cheeks: { cx: 70, cy: 95, r: 18, label: 'L Cheek' },
   314→    chin: { cx: 100, cy: 125, r: 12, label: 'Chin' },
   315→    jawline: { cx: 55, cy: 125, r: 14, label: 'Jawline' },
   316→  }
   317→
   318→  return (
   319→    <div className="flex flex-col items-center gap-3">
   320→      <svg viewBox="0 0 200 170" className="w-full max-w-[260px] h-auto">
   321→        {/* Face shape */}
   322→        <ellipse
   323→          cx="100"
   324→          cy="90"
   325→          rx="55"
   326→          ry="75"
   327→          fill="url(#faceGrad)"
   328→          stroke="oklch(0.8 0.05 30)"
   329→          strokeWidth="2"
   330→        />
   331→        <defs>
   332→          <radialGradient id="faceGrad" cx="50%" cy="40%">
   333→            <stop offset="0%" stopColor="oklch(0.95 0.04 30)" />
   334→            <stop offset="100%" stopColor="oklch(0.88 0.05 30)" />
   335→          </radialGradient>
   336→        </defs>
   337→        {/* Hair hint */}
   338→        <path d="M 45 35 Q 100 5 155 35 L 155 25 Q 100 -5 45 25 Z" fill="oklch(0.35 0.08 30)" opacity="0.85" />
   339→
   340→        {/* Clickable areas */}
   341→        {(['forehead', 'tzone', 'nose', 'cheeks', 'chin', 'jawline'] as FaceArea[]).map((area) => {
   342→          const meta = areas[area]
   343→          const sel = isSelected(area)
   344→          return (
   345→            <g key={area} onClick={() => onToggle(area)} className="cursor-pointer">
   346→              {/* Right cheek mirror */}
   347→              {area === 'cheeks' && (
   348→                <circle
   349→                  cx={200 - meta.cx}
   350→                  cy={meta.cy}
   351→                  r={meta.r}
   352→                  fill={sel ? 'oklch(0.7 0.18 350 / 0.5)' : 'oklch(0.7 0.05 30 / 0.15)'}
   353→                  stroke={sel ? 'oklch(0.55 0.2 350)' : 'oklch(0.6 0.05 30)'}
   354→                  strokeWidth="1.5"
   355→                  className="transition-all"
   356→                />
   357→              )}
   358→              {area === 'jawline' && (
   359→                <circle
   360→                  cx={200 - meta.cx}
   361→                  cy={meta.cy}
   362→                  r={meta.r}
   363→                  fill={sel ? 'oklch(0.7 0.18 350 / 0.5)' : 'oklch(0.7 0.05 30 / 0.15)'}
   364→                  stroke={sel ? 'oklch(0.55 0.2 350)' : 'oklch(0.6 0.05 30)'}
   365→                  strokeWidth="1.5"
   366→                  className="transition-all"
   367→                />
   368→              )}
   369→              <circle
   370→                cx={meta.cx}
   371→                cy={meta.cy}
   372→                r={meta.r}
   373→                fill={sel ? 'oklch(0.7 0.18 350 / 0.5)' : 'oklch(0.7 0.05 30 / 0.15)'}
   374→                stroke={sel ? 'oklch(0.55 0.2 350)' : 'oklch(0.6 0.05 30)'}
   375→                strokeWidth={sel ? 2.5 : 1.5}
   376→                className="transition-all"
   377→              />
   378→              {sel && (
   379→                <text
   380→                  x={meta.cx}
   381→                  y={meta.cy + 4}
   382→                  textAnchor="middle"
   383→                  fontSize="14"
   384→                  fill="oklch(0.4 0.15 350)"
   385→                  fontWeight="bold"
   386→                >
   387→                  ✓
   388→                </text>
   389→              )}
   390→            </g>
   391→          )
   392→        })}
   393→      </svg>
   394→      <div className="flex flex-wrap gap-1.5 justify-center max-w-[280px]">
   395→        {FACE_AREAS.map((area) => (
   396→          <button
   397→            key={area.id}
   398→            onClick={() => onToggle(area.id)}
   399→            className={cn(
   400→              'text-[11px] px-2.5 py-1 rounded-full border transition-all',
   401→              isSelected(area.id)
   402→                ? 'bg-rose-500 text-white border-rose-500'
   403→                : 'bg-muted/50 text-muted-foreground border-border hover:border-rose-300'
   404→            )}
   405→          >
   406→            {area.label}
   407→          </button>
   408→        ))}
   409→      </div>
   410→    </div>
   411→  )
   412→}

   475→          ) : (
   476→            <div className="flex flex-col items-center justify-center py-8 px-4 text-center">
   477→              <div className="flex h-14 w-14 items-center justify-center rounded-2xl bg-muted mb-3">

   490→      <Card className="lg:col-span-2">
   491→        <CardHeader className="pb-3">
   492→          <CardTitle className="text-base">Score Breakdown</CardTitle>
   493→          <CardDescription>Key skin health indicators</CardDescription>
   494→        </CardHeader>
   495→        <CardContent className="space-y-4 pt-2">
   496→          {hasScore ? (
   497→            <>
