   239→
   240→function PlanCard({ plan, billingCycle, onChoosePlan }: { plan: Plan; billingCycle: BillingCycle; onChoosePlan: (plan: Plan) => void }) {
   241→  const hasPremium = useAppStore((s) => s.hasPremium())
   242→  const price = billingCycle === 'monthly' ? plan.monthlyPrice : plan.yearlyPrice
   243→  const isFree = plan.id === 'free'
   244→  // User is on "free" if they don't have premium; otherwise they're on "premium"
   245→  const isCurrentPlan = hasPremium ? plan.id === 'premium' : plan.id === 'free'
   246→  const yearlyPerMonth = plan.yearlyPrice / 12
   247→  const savings = plan.monthlyPrice > 0 && plan.yearlyPrice > 0
   248→    ? Math.round(((plan.monthlyPrice * 12 - plan.yearlyPrice) / (plan.monthlyPrice * 12)) * 100)
   249→    : 0
   250→
   251→  return (
   252→    <motion.div
   253→      initial={{ opacity: 0, y: 20 }}
   254→      whileInView={{ opacity: 1, y: 0 }}
   255→      viewport={{ once: true }}
   256→      transition={{ duration: 0.4 }}
   257→      whileHover={{ y: -4 }}
   258→      className={cn('relative', plan.isPopular && 'lg:-mt-4 lg:mb-0')}
   259→    >
   260→      <Card className={cn(
   261→        'relative h-full overflow-hidden transition-all',
   262→        plan.isPopular
   263→          ? 'border-amber-400 dark:border-amber-600 shadow-xl shadow-amber-500/20 ring-2 ring-amber-400/50'
   264→          : plan.borderColor
   265→      )}>
   266→        {/* Popular badge */}
   267→        {plan.isPopular && (
   268→          <div className="absolute top-0 left-1/2 -translate-x-1/2 -translate-y-1/2 z-10">
   269→            <div className="flex items-center gap-1.5 rounded-full bg-gradient-to-r from-amber-400 via-yellow-500 to-orange-500 px-4 py-1.5 text-xs font-bold text-white shadow-lg">
   270→              <Sparkles className="h-3 w-3" />
   271→              {plan.badgeText}
   272→            </div>
   273→          </div>
   274→        )}
   275→
   276→        {/* Gradient top */}
   277→        <div className={cn('h-2 bg-gradient-to-r', plan.gradient)} />
   278→
   279→        <CardHeader className="pt-5 pb-3">
   280→          <div className="flex items-center justify-between">
   281→            <div className={cn('flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br text-white shadow-md', plan.gradient)}>
   282→              <plan.icon className="h-5 w-5" />
   283→            </div>
   284→            {savings > 0 && billingCycle === 'yearly' && (
   285→              <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 gap-1">
   286→                <Gift className="h-3 w-3" /> Save {savings}%
   287→              </Badge>
   288→            )}
   289→          </div>
   290→          <div className="mt-2">
   291→            <CardTitle className="text-xl">{plan.name}</CardTitle>
   292→            <CardDescription>{plan.tagline}</CardDescription>
   293→          </div>
   294→        </CardHeader>
   295→
   296→        <CardContent className="pb-3">
   297→          {/* Price */}
   298→          <div className="mb-4">
   299→            {isFree ? (
   300→              <div className="flex items-baseline gap-1">
   301→                <span className="text-4xl font-bold">₹0</span>
   302→                <span className="text-sm text-muted-foreground">forever</span>
   303→              </div>
   304→            ) : (
   305→              <div>
   306→                <div className="flex items-baseline gap-1">
   307→                  <span className="text-4xl font-bold">₹{price.toLocaleString('en-IN')}</span>
   308→                  <span className="text-sm text-muted-foreground">/{billingCycle === 'monthly' ? 'month' : 'year'}</span>
   309→                </div>
   310→                {billingCycle === 'yearly' && (
   311→                  <p className="text-xs text-emerald-600 dark:text-emerald-400 mt-1">
   312→                    Just ₹{yearlyPerMonth.toFixed(0)}/month · billed annually
   313→                  </p>
   314→                )}
   315→              </div>
   316→            )}
   317→          </div>
   318→
   319→          <Separator className="mb-4" />
   320→
   321→          {/* Features */}
   322→          <ul className="space-y-2.5">
   323→            {plan.features.map((feature) => (
   324→              <li key={feature.text} className="flex items-start gap-2.5">
   325→                <div className={cn(
   326→                  'flex h-4 w-4 items-center justify-center rounded-full shrink-0 mt-0.5',
   327→                  feature.included
   328→                    ? plan.isPopular ? 'bg-amber-100 dark:bg-amber-950/50' : 'bg-emerald-100 dark:bg-emerald-950/40'
   329→                    : 'bg-muted/50'
   330→                )}>
   331→                  {feature.included ? (
   332→                    <Check className={cn('h-3 w-3', plan.isPopular ? 'text-amber-600 dark:text-amber-400' : 'text-emerald-600 dark:text-emerald-400')} />
   333→                  ) : (
   334→                    <X className="h-2.5 w-2.5 text-muted-foreground" />
   335→                  )}
   336→                </div>
   337→                <span className={cn(
   338→                  'text-sm leading-snug',
   339→                  feature.included ? 'text-foreground' : 'text-muted-foreground/60 line-through'
   340→                )}>
   341→                  {feature.text}
   342→                </span>
   343→              </li>
   344→            ))}
   345→          </ul>
   346→        </CardContent>
   347→
   348→        <CardFooter className="pt-0">
   349→          <Button
   350→            className={cn(
   351→              'w-full',
   352→              isCurrentPlan
   353→                ? 'bg-muted text-muted-foreground'
   354→                : plan.isPopular
   355→                ? 'bg-gradient-to-r from-amber-500 via-yellow-500 to-orange-500 hover:opacity-90 text-white shadow-md shadow-amber-500/30'
   356→                : plan.id === 'plus'
   357→                ? 'bg-gradient-to-r from-purple-500 via-fuchsia-500 to-pink-500 hover:opacity-90 text-white shadow-md shadow-fuchsia-500/30'
   358→                : 'bg-foreground text-background hover:bg-foreground/90'
   359→            )}
   360→            disabled={isCurrentPlan}
   361→            onClick={() => {
   362→              if (isCurrentPlan) return
   363→              onChoosePlan(plan)
   364→            }}
   365→          >
   366→            {isCurrentPlan ? (
   367→              <>
   368→                <Check className="h-4 w-4 mr-1.5" /> Current Plan
   369→              </>
   370→            ) : (
   371→              <>
   372→                {plan.cta}
   373→                <ArrowRight className="h-4 w-4 ml-1.5" />
   374→              </>
   375→            )}
   376→          </Button>
   377→        </CardFooter>
   378→      </Card>
   379→    </motion.div>
   380→  )
   381→}
