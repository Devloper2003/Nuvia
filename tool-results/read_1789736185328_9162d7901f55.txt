   512→function CircularGauge({ value, size = 180 }: { value: number; size?: number }) {
   513→  const strokeWidth = 14
   514→  const radius = (size - strokeWidth) / 2
   515→  const circumference = 2 * Math.PI * radius
   516→  const progress = (value / 100) * circumference
   517→  const center = size / 2
   518→
   519→  const getColorZone = (val: number) => {
   520→    if (val <= 30) return { color: '#22c55e', label: 'Low Risk', bg: 'bg-green-50 dark:bg-green-950/20' }
   521→    if (val <= 55) return { color: '#eab308', label: 'Moderate Risk', bg: 'bg-yellow-50 dark:bg-yellow-950/20' }
   522→    if (val <= 75) return { color: '#f97316', label: 'High Risk', bg: 'bg-orange-50 dark:bg-orange-950/20' }
   523→    return { color: '#ef4444', label: 'Very High Risk', bg: 'bg-red-50 dark:bg-red-950/20' }
   524→  }
   525→
   526→  const zone = getColorZone(value)
   527→
   528→  return (
   529→    <div className="flex flex-col items-center gap-3">
   530→      <div className={`relative rounded-2xl p-6 ${zone.bg}`}>
   531→        <svg width={size} height={size} className="transform -rotate-90">
   532→          {/* Background circle */}
   533→          <circle
   534→            cx={center}
   535→            cy={center}
   536→            r={radius}
   537→            fill="none"
   538→            stroke="currentColor"
   539→            strokeWidth={strokeWidth}
   540→            className="text-muted/30"
   541→          />
   542→          {/* Progress arc */}
   543→          <motion.circle
   544→            cx={center}
   545→            cy={center}
   546→            r={radius}
   547→            fill="none"
   548→            stroke={zone.color}
   549→            strokeWidth={strokeWidth}
   550→            strokeLinecap="round"
   551→            strokeDasharray={circumference}
   552→            initial={{ strokeDashoffset: circumference }}
   553→            animate={{ strokeDashoffset: circumference - progress }}
   554→            transition={{ duration: 1.5, ease: 'easeOut' }}
   555→          />
   556→        </svg>
   557→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   558→          <motion.span
   559→            initial={{ opacity: 0, scale: 0.5 }}
   560→            animate={{ opacity: 1, scale: 1 }}
   561→            transition={{ delay: 0.5 }}
   562→            className="text-4xl font-bold"
   563→            style={{ color: zone.color }}
   564→          >
   565→            {value}
   566→          </motion.span>
   567→          <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider mt-0.5">
   568→            / 100
   569→          </span>
   570→        </div>
   571→      </div>
   572→      <Badge
   573→        className="text-xs font-semibold px-3 py-1"

   575→        variant="outline"
   576→      >
   577→        {zone.label}
   578→      </Badge>
   579→    </div>

   585→function ProgressRing({ value, max, size = 80, color = '#ec4899' }: { value: number; max: number; size?: number; color?: string }) {

   587→  const radius = (size - strokeWidth) / 2

   595→        <circle cx={center} cy={center} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/30" />

   608→        />
