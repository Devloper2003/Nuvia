   131→// ─── Helper: debounce ─────────────────────────────────────────────────────────
   132→function debounce<T extends (...args: any[]) => void>(fn: T, delay: number) {
   133→  let t: ReturnType<typeof setTimeout>
   134→  return (...args: Parameters<T>) => {
   135→    clearTimeout(t)
   136→    t = setTimeout(() => fn(...args), delay)
   137→  }
   138→}

   362→function getAvatarGradient(name: string): string {
   363→  let hash = 0
   364→  for (let i = 0; i < name.length; i++) {
   365→    hash = name.charCodeAt(i) + ((hash << 5) - hash)
   366→  }
   367→  return AVATAR_GRADIENTS[Math.abs(hash) % AVATAR_GRADIENTS.length]
   368→}

   370→function getInitials(name: string): string {
   371→  const parts = name.replace(/^Dr\.\s*/i, '').split(' ')
   372→  return (parts[0]?.[0] ?? '') + (parts[1]?.[0] ?? '')
   373→}

   375→// ─── Sub-component: Star Rating ───────────────────────────────────────────────
   376→
   377→function StarRating({ rating }: { rating: number }) {
   378→  const full = Math.floor(rating)
   379→  const hasHalf = rating - full >= 0.5
   380→  return (
   381→    <div className="flex items-center gap-0.5" aria-label={`Rating ${rating} out of 5`}>
   382→      {Array.from({ length: 5 }).map((_, i) => {
   383→        const isFull = i < full
   384→        const isHalf = i === full && hasHalf
   385→        return (
   386→          <Star
   387→            key={i}
   388→            className={cn(
   389→              'h-3.5 w-3.5',
   390→              isFull || isHalf
   391→                ? 'fill-amber-400 text-amber-400'
   392→                : 'fill-muted text-muted-foreground/40'
   393→            )}
   394→          />
   395→        )
   396→      })}
   397→    </div>
   398→  )
   399→}

   548→  // Derived: filtered + sorted doctors
