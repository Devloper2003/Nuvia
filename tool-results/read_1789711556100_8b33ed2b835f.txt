    79→// ── Helpers ─────────────────────────────────────────────────────────────────
    80→function TrendIcon({ current, previous }: { current: number; previous: number }) {
    81→  const diff = current - previous
    82→  if (diff > 2) return <TrendingUp className="size-4 text-emerald-500" />
    83→  if (diff < -2) return <TrendingDown className="size-4 text-red-400" />
    84→  return <Minus className="size-4 text-muted-foreground" />
    85→}
    86→
    87→function TrendLabel({ current, previous, suffix = '' }: { current: number; previous: number; suffix?: string }) {
    88→  const diff = current - previous
    89→  const pct = previous !== 0 ? Math.round((diff / previous) * 100) : 0
    90→  if (diff > 2)
    91→    return (
    92→      <span className="text-xs text-emerald-600 dark:text-emerald-400 font-medium">
    93→        +{pct}%{suffix}
    94→      </span>
    95→    )
    96→  if (diff < -2)
    97→    return (
    98→      <span className="text-xs text-red-500 font-medium">
    99→        {pct}%{suffix}
   100→      </span>
   101→    )
   102→  return (
   103→    <span className="text-xs text-muted-foreground font-medium">No change{suffix}</span>
   104→  )
   105→}
