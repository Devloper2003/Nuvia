    75→// ── Helpers ─────────────────────────────────────────────────────────────────
    76→function TrendIcon({ current, previous }: { current: number; previous: number }) {
    77→  const diff = current - previous
    78→  if (diff > 2) return <TrendingUp className="size-4 text-emerald-500" />
    79→  if (diff < -2) return <TrendingDown className="size-4 text-red-400" />
    80→  return <Minus className="size-4 text-muted-foreground" />
    81→}
    82→
    83→function TrendLabel({ current, previous, suffix = '' }: { current: number; previous: number; suffix?: string }) {
    84→  const diff = current - previous
    85→  const pct = previous !== 0 ? Math.round((diff / previous) * 100) : 0
    86→  if (diff > 2)
    87→    return (
    88→      <span className="text-xs text-emerald-600 dark:text-emerald-400 font-medium">
    89→        +{pct}%{suffix}
    90→      </span>
    91→    )
    92→  if (diff < -2)
    93→    return (
    94→      <span className="text-xs text-red-500 font-medium">
    95→        {pct}%{suffix}
    96→      </span>
    97→    )
    98→  return (
    99→    <span className="text-xs text-muted-foreground font-medium">No change{suffix}</span>
   100→  )
   101→}
