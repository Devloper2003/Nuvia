    59→// Generates a TYPICAL (textbook) hormone curve for educational reference.
    60→// This is not presented as the user's own data — only as an illustrative
    61→// reference chart labeled "Typical hormone pattern".
    62→function generateTypicalCycleData(cycleLength: number): CycleDayData[] {
    63→  const data: CycleDayData[] = []
    64→  const ovulationDay = cycleLength - 14
    65→  const periodEnd = Math.min(5, Math.floor(cycleLength / 5))
    66→  for (let d = 1; d <= cycleLength; d++) {
    67→    let estrogen = 0
    68→    let progesterone = 0
    69→    let lh = 0
    70→    let fsh = 0
    71→    let phase = ''
    72→
    73→    if (d <= periodEnd) {
    74→      phase = 'Menstrual'
    75→      estrogen = 25
    76→      progesterone = 8
    77→      lh = 6
    78→      fsh = 10
    79→    } else if (d < ovulationDay - 1) {
    80→      phase = 'Follicular'
    81→      const progress = (d - periodEnd) / (ovulationDay - 1 - periodEnd)
    82→      estrogen = 30 + progress * 110
    83→      progesterone = 8
    84→      lh = 6
    85→      fsh = 10 + progress * 4
    86→    } else if (d >= ovulationDay - 1 && d <= ovulationDay + 1) {
    87→      phase = 'Ovulation'
    88→      estrogen = 145
    89→      progesterone = 15
    90→      lh = d === ovulationDay ? 90 : d === ovulationDay - 1 ? 50 : 30
    91→      fsh = 14
    92→    } else {
    93→      phase = 'Luteal'
    94→      const lutealProgress = (d - (ovulationDay + 1)) / (cycleLength - ovulationDay - 1)
    95→      estrogen = 130 - lutealProgress * 80
    96→      progesterone = 15 + lutealProgress * 95 - (lutealProgress > 0.6 ? lutealProgress * 50 : 0)
    97→      lh = 6
    98→      fsh = 6
    99→    }
   100→
   101→    data.push({
   102→      day: d,
   103→      phase,
   104→      estrogen: Math.round(estrogen * 10) / 10,
   105→      progesterone: Math.round(progesterone * 10) / 10,
   106→      lh: Math.round(lh * 10) / 10,
   107→      fsh: Math.round(fsh * 10) / 10,
   108→    })
   109→  }
   110→  return data
   111→}

   261→// ─── Cycle data type ────────────────────────────────────────────────────────

   263→interface CycleEntry {
   264→  id: string
   265→  startDate: string
   266→  endDate?: string | null
   267→  cycleLength: number
   268→  periodLength: number
   269→  ovulationDate?: string | null
   270→}
   271→
   272→function deriveCycleDay(startISO: string, cycleLength: number): number | null {
   273→  const start = new Date(startISO)
   274→  start.setHours(0, 0, 0, 0)
   275→  const today = new Date()
   276→  today.setHours(0, 0, 0, 0)
   277→  const diff = Math.floor((today.getTime() - start.getTime()) / (1000 * 60 * 60 * 24)) + 1
   278→  if (diff < 1) return null
   279→  return ((diff - 1) % cycleLength) + 1
   280→}
   281→
   282→function getPhaseForCycleDay(day: number, cycleLength: number): string {
   283→  const ovulationDay = cycleLength - 14
   284→  const periodEnd = Math.min(5, Math.floor(cycleLength / 5))
   285→  if (day <= periodEnd) return 'Menstrual'
   286→  if (day < ovulationDay - 1) return 'Follicular'
   287→  if (day >= ovulationDay - 1 && day <= ovulationDay + 1) return 'Ovulation'
   288→  return 'Luteal'
   289→}

   554→
   555→function HormoneLog({ userId }: { userId: string | undefined }) {
   556→  const [selectedSymptoms, setSelectedSymptoms] = useState<string[]>([])
   557→  const [notes, setNotes] = useState('')
   558→  const [saved, setSaved] = useState(false)
   559→  const [saving, setSaving] = useState(false)
   560→
   561→  const toggleSymptom = (id: string) => {
   562→    setSelectedSymptoms((prev) => (prev.includes(id) ? prev.filter((s) => s !== id) : [...prev, id]))
   563→    setSaved(false)
   564→  }
   565→
   566→  const handleSave = async () => {
   567→    if (!userId) return
   568→    setSaving(true)
   569→    try {
   570→      const today = new Date().toISOString().split('T')[0]
   571→      await Promise.all(
   572→        selectedSymptoms.map((category) =>
   573→          fetch('/api/symptoms', {
   574→            method: 'POST',
   575→            headers: { 'Content-Type': 'application/json' },
   576→            body: JSON.stringify({ userId, category, severity: 3, date: today, notes: notes || undefined }),
   577→          })
   578→        )
   579→      )
   580→      setSaved(true)
   581→      setTimeout(() => {
   582→        setSaved(false)
   583→        setSelectedSymptoms([])
   584→        setNotes('')
   585→      }, 2500)
   586→    } catch (e) {
   587→      console.error('Failed to save hormone symptoms:', e)
   588→      toast.error('Could not save. Please try again.')
   589→    } finally {
   590→      setSaving(false)
   591→    }
   592→  }
   593→
   594→  return (
   595→    <motion.div variants={itemVariants}>
   596→      <Card className="border-0 bg-white/60 dark:bg-gray-900/60 backdrop-blur-xl shadow-lg">
   597→        <CardHeader className="pb-3">
   598→          <CardTitle className="flex items-center gap-2 text-lg">
   599→            <Activity className="h-5 w-5 text-purple-500" />
   600→            Hormone Symptom Log
   601→          </CardTitle>
   602→          <p className="text-xs text-muted-foreground">Track symptoms that correlate with your hormonal changes</p>
   603→        </CardHeader>
   604→        <CardContent className="space-y-4">
   605→          <div className="space-y-1.5">
   606→            <span className="text-xs font-medium text-muted-foreground">Select symptoms you&apos;re experiencing</span>
   607→            <div className="flex flex-wrap gap-2">
   608→              {symptomCategories.map((symptom) => {
   609→                const isSelected = selectedSymptoms.includes(symptom.id)
   610→                return (
   611→                  <motion.button
   612→                    key={symptom.id}
   613→                    whileTap={{ scale: 0.95 }}
   614→                    onClick={() => toggleSymptom(symptom.id)}
   615→                    className={`inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium transition-all duration-200 border ${
   616→                      isSelected
   617→                        ? 'bg-purple-100 border-purple-300 text-purple-700 dark:bg-purple-900/40 dark:border-purple-700 dark:text-purple-300 shadow-sm'
   618→                        : 'bg-background border-border/50 text-muted-foreground hover:border-purple-300 hover:text-purple-600 dark:hover:border-purple-700 dark:hover:text-purple-400'
   619→                    }`}
   620→                  >
   621→                    <symptom.icon className="h-3 w-3" />
   622→                    {symptom.label}
   623→                    {isSelected && <CheckCircle2 className="h-3 w-3" />}
   624→                  </motion.button>
   625→                )
   626→              })}
   627→            </div>
   628→          </div>
   629→
   630→          <div className="space-y-1.5">
   631→            <span className="text-xs font-medium text-muted-foreground">Additional notes</span>
   632→            <textarea
   633→              className="w-full rounded-xl border border-border/50 bg-background px-3 py-2.5 text-sm text-foreground placeholder:text-muted-foreground/60 focus:outline-none focus:ring-2 focus:ring-purple-500/30 focus:border-purple-400 resize-none transition-all"
   634→              rows={3}
   635→              placeholder="Describe how you're feeling, any patterns you notice..."
   636→              value={notes}
   637→              onChange={(e) => {
   638→                setNotes(e.target.value)
   639→                setSaved(false)
   640→              }}
   641→            />
   642→          </div>
   643→
   644→          <div className="flex items-center gap-3">
   645→            <Button
   646→              onClick={handleSave}
   647→              disabled={saving || !userId || (selectedSymptoms.length === 0 && !notes)}
   648→              className="bg-gradient-to-r from-purple-600 to-rose-600 hover:from-purple-700 hover:to-rose-700 text-white shadow-lg shadow-purple-500/20 transition-all duration-200"
   649→            >
   650→              {saved ? (
   651→                <>
   652→                  <CheckCircle2 className="h-4 w-4 mr-1.5" />
   653→                  Saved!
   654→                </>
   655→              ) : (
   656→                <>
   657→                  <Sparkles className="h-4 w-4 mr-1.5" />
   658→                  Log Symptoms
   659→                </>
   660→              )}
   661→            </Button>
   662→            {saved && (
   663→              <motion.p
   664→                initial={{ opacity: 0, x: -10 }}
   665→                animate={{ opacity: 1, x: 0 }}
   666→                className="text-xs text-emerald-600 dark:text-emerald-400"
   667→              >
   668→                Symptoms logged successfully — AI will correlate with your hormone data
   669→              </motion.p>
   670→            )}
   671→            {selectedSymptoms.length > 0 && (
   672→              <span className="text-xs text-muted-foreground">
   673→                {selectedSymptoms.length} symptom{selectedSymptoms.length !== 1 ? 's' : ''} selected
   674→              </span>
   675→            )}
   676→          </div>
   677→        </CardContent>
   678→      </Card>
   679→    </motion.div>
   680→  )
   681→}
