   136→function SettingsSection({

   162→          <div className="flex items-center gap-3">

   164→              <Icon className="h-4 w-4" />
   165→            </div>
   166→            <div>

   168→              {description && <CardDescription className="text-xs">{description}</CardDescription>}

   171→        </CardHeader>
